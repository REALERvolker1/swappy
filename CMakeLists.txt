cmake_minimum_required(VERSION 3.23)
project(swappy VERSION 1.8.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
include(GNUInstallDirs)
include(FetchContent)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0 gio-2.0 gio-unix-2.0)
pkg_check_modules(PIXBUF REQUIRED IMPORTED_TARGET gdk-pixbuf-2.0)
pkg_check_modules(CAIRO REQUIRED IMPORTED_TARGET cairo pangocairo)
pkg_check_modules(TESSERACT REQUIRED IMPORTED_TARGET tesseract)
find_package(raven CONFIG QUIET)
if(NOT raven_FOUND)
 set(RAVEN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(RAVEN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 set(RAVEN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
 set(RAVEN_BUILD_WEB OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(raven GIT_REPOSITORY https://github.com/oscar-soirey/Raven.git GIT_TAG c4410662723bf0eec633a36a7c2e4d69490325cf)
 FetchContent_MakeAvailable(raven)
endif()
add_executable(swappy src/main.c src/algebra.c src/application.c src/box.c src/config.c src/clipboard.c src/file.c src/ocr.c src/paint.c src/pixbuf.c src/render.c src/selection.c src/util.c)
target_include_directories(swappy PRIVATE include)
target_compile_definitions(swappy PRIVATE SWAPPY_VERSION="1.8.0")
target_link_libraries(swappy PRIVATE raven::raven PkgConfig::GLIB PkgConfig::PIXBUF PkgConfig::CAIRO PkgConfig::TESSERACT m)
install(TARGETS swappy RUNTIME DESTINATION bin)
